        <script>
            let currentBSlide = 0;
            const bSlides = document.querySelectorAll('.brochure-slide');
            const bDots = document.querySelectorAll('.slider-dot');
            let bSlideInterval;

            function showBSlide(index) {
                bSlides.forEach(s => s.classList.remove('active'));
                bDots.forEach(d => d.classList.remove('active'));
                if(bSlides[index]) bSlides[index].classList.add('active');
                if(bDots[index]) bDots[index].classList.add('active');
                currentBSlide = index;
            }

            function nextBSlide() {
                if(bSlides.length === 0) return;
                let next = (currentBSlide + 1) % bSlides.length;
                showBSlide(next);
            }

            function goToSlide(index) {
                showBSlide(index);
                clearInterval(bSlideInterval);
                bSlideInterval = setInterval(nextBSlide, 4000);
            }

            document.addEventListener('DOMContentLoaded', () => {
                if(bSlides.length > 0) {
                    bSlideInterval = setInterval(nextBSlide, 4000);
                }
            });
        </script>
